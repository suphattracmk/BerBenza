โค้ดสำหรับวางใน Google Apps Script 
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    sheet.appendRow([
      new Date(),
      data.name,
      data.phone,
      data.province,
      data.hobbies,
      data.weekend,
      data.idealTemp,
      data.prizeWon
    ]);
    return ContentService.createTextOutput(JSON.stringify({"result": "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({"result": "error", "error": err.message}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
  1. สร้าง Google Sheet ใหม่ใน Google Drive ของคุณ
  2. ไปที่เมนู ส่วนขยาย (Extensions) > Apps Script
  3. วางโค้ดด้านบน แล้วกด การทำให้ใช้งานได้ (Deploy) > การทำให้ใช้งานได้ใหม่ (New deployment)
  4. เลือกประเภท เว็บแอป (Web app) และตั้งค่า ผู้มีสิทธิ์เข้าถึง (Who has access) เป็น ทุกคน (Anyone)
