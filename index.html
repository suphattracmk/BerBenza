/** * ============================================================================== * Project: Carrier Cool Quest Survey - Google Sheets Webhook Backend * Description: ระบบรับข้อมูลแบบสอบถามและผลหมุนวงล้อรางวัล บันทึกลง Google Sheets * * ตารางระบุตำแหน่งหัวคอลัมน์ (Column Headers Reference - แถวที่ 1): * ------------------------------------------------------------------------------ * | ช่อง (Cell) | ชื่อหัวคอลัมน์ (Header Name) | รายละเอียดข้อมูลที่เก็บ | * |-------------|-----------------------------|-----------------------------------------| * | A1 | วัน-เวลาบันทึก | วันที่และเวลาส่งข้อมูล (GMT+7) | * | B1 | ชื่อ - นามสกุล | ชื่อและนามสกุลของผู้ร่วมกิจกรรม | * | C1 | เบอร์โทรศัพท์ | เบอร์มือถือ 10 หลัก (มี ' นำหน้าป้องกัน 0 หาย)| * | D1 | จังหวัด | จังหวัดที่พักอาศัย | * | E1 | กิจกรรมวันหยุดที่ชอบ | กิจกรรมที่เลือกทำในวันหยุด | * | F1 | งานอดิเรก | งานอดิเรกที่เลือก (คั่นด้วยเครื่องหมายจุลภาค) | * | G1 | อุณหภูมิแอร์ในฝัน | อุณหภูมิองศาเซลเซียส เช่น 25°C | * | H1 | ของรางวัลที่ได้รับ | ผลรางวัลที่ได้จากการหมุนวงล้อ | * | I1 | รหัสคูปอง | รหัสคูปองรางวัล เช่น CR-AB12-COOL | * ------------------------------------------------------------------------------ * Version: 3.0 (รองรับการกดรันตั้งค่าหัวตารางล่วงหน้า setupHeadersManually ได้ทันที) * ============================================================================== */ // ชื่อแท็บของชีตที่จะบันทึกข้อมูล var SHEET_NAME = "Carrier_Survey_Data"; // การตั้งค่าสีและสไตล์หัวตารางตามธีมแบรนด์ Carrier var THEME = { headerBg: "#002B66", // Carrier Navy Blue headerText: "#FFFFFF", // White fontSize: 11, fontFamily: "Prompt" }; /** * ------------------------------------------------------------------------------ * [ฟังก์ชันสำหรับกด รัน (Run) ด้วยตนเองเพื่อทดสอบหรือสร้างตารางล่วงหน้า] * คุณสามารถเลือกฟังก์ชันนี้ใน Apps Script แล้วกดปุ่ม "เรียกใช้ (Run)" ได้ทันที! * ------------------------------------------------------------------------------ */ function setupHeadersManually() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); } initHeaderRow(sheet); Logger.log("✅ สร้างและจัดรูปแบบหัวตาราง Carrier เรียบร้อยแล้วในแท็บ: " + SHEET_NAME); } /** * ฟังก์ชันช่วยสร้างและจัดสไตล์หัวตาราง (A1 ถึง I1) */ function initHeaderRow(sheet) { var headers = [ "วัน-เวลาบันทึก", // ช่อง A1 "ชื่อ - นามสกุล", // ช่อง B1 "เบอร์โทรศัพท์", // ช่อง C1 "จังหวัด", // ช่อง D1 "กิจกรรมวันหยุดที่ชอบ", // ช่อง E1 "งานอดิเรก", // ช่อง F1 "อุณหภูมิแอร์ในฝัน", // ช่อง G1 "ของรางวัลที่ได้รับ", // ช่อง H1 "รหัสคูปอง" // ช่อง I1 ]; // เขียนทับแถวที่ 1 เสมอเพื่อให้มั่นใจว่าหัวตารางถูกต้องครบถ้วน sheet.getRange(1, 1, 1, headers.length).setValues([headers]); // จัดแต่งสไตล์หัวตาราง Carrier Navy var headerRange = sheet.getRange(1, 1, 1, headers.length); headerRange.setFontWeight("bold") .setBackground(THEME.headerBg) .setFontColor(THEME.headerText) .setFontSize(THEME.fontSize) .setHorizontalAlignment("center") .setVerticalAlignment("middle"); // ปรับความสูงแถวหัวตารางและตรึงแถวที่ 1 ไว้ sheet.setRowHeight(1, 40); sheet.setFrozenRows(1); // ปรับความกว้างคอลัมน์ให้อ่านง่าย sheet.setColumnWidth(1, 170); // วันที่ sheet.setColumnWidth(2, 180); // ชื่อ sheet.setColumnWidth(3, 140); // เบอร์โทร sheet.setColumnWidth(4, 150); // จังหวัด sheet.setColumnWidth(5, 220); // กิจกรรมวันหยุด sheet.setColumnWidth(6, 240); // งานอดิเรก sheet.setColumnWidth(7, 130); // อุณหภูมิ sheet.setColumnWidth(8, 220); // ของรางวัล sheet.setColumnWidth(9, 160); // รหัสคูปอง } /** * ฟังก์ชัน doGet: ใช้สำหรับทดสอบว่า Web App ทำงานปกติหรือไม่เมื่อเปิดผ่านเบราว์เซอร์ */ function doGet(e) { var output = { status: "online", service: "Carrier Cool Quest Webhook", message: "ระบบ Carrier Survey Webhook พร้อมรับข้อมูลแล้ว!", timestamp: Utilities.formatDate(new Date(), "GMT+7", "dd/MM/yyyy HH:mm:ss") }; return ContentService.createTextOutput(JSON.stringify(output)) .setMimeType(ContentService.MimeType.JSON); } /** * ฟังก์ชัน doPost: รับข้อมูลแบบสอบถามจากหน้าเว็บแล้วบันทึกลงชีต */ function doPost(e) { // สร้าง Script Lock เพื่อป้องกันกรณีที่ลูกค้าส่งข้อมูลพร้อมกันหลายคนในเสี้ยววินาที var lock = LockService.getScriptLock(); // ให้คิวรอนานสูงสุด 20 วินาที var hasLock = lock.tryLock(20000); if (!hasLock) { return ContentService.createTextOutput(JSON.stringify({ result: "error", message: "Server is busy. Please try again." })).setMimeType(ContentService.MimeType.JSON); } try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(SHEET_NAME); // หากยังไม่มีแท็บที่กำหนด ให้สร้างแท็บใหม่ขึ้นมาโดยอัตโนมัติ if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); } // ตรวจสอบว่ามีหัวตารางหรือยัง หากยังไม่มี ให้สร้างหัวตาราง if (sheet.getLastRow() === 0) { initHeaderRow(sheet); } // แยกอ่านข้อมูล JSON ที่ส่งมาจากหน้าเว็บ var data = {}; if (e && e.postData && e.postData.contents) { try { data = JSON.parse(e.postData.contents); } catch (err) { data = e.parameter || {}; } } else if (e && e.parameter) { data = e.parameter; } // กำหนดรูปแบบวัน-เวลาของประเทศไทย (GMT+7) var timestamp = Utilities.formatDate(new Date(), "GMT+7", "dd/MM/yyyy HH:mm:ss"); // จัดการเบอร์โทรศัพท์: ใส่เครื่องหมาย ' นำหน้า ป้องกันไม่ให้ Google Sheets ตัดเลข 0 ด้านหน้าทิ้ง var rawPhone = data.phone ? String(data.phone).trim() : "-"; var formattedPhone = rawPhone; if (formattedPhone !== "-" && !formattedPhone.startsWith("'")) { formattedPhone = "'" + formattedPhone; } // จัดเตรียมแถวข้อมูลที่จะบันทึก var newRow = [ timestamp, data.name || "-", formattedPhone, data.province || "-", data.weekend || "-", data.hobbies || "-", data.idealTemp || "-", data.prizeWon || "-", data.couponCode || "-" ]; sheet.appendRow(newRow); // จัดกึ่งกลางสำหรับคอลัมน์สำคัญ (วันที่, เบอร์โทร, จังหวัด, อุณหภูมิ, รหัสคูปอง) var lastRow = sheet.getLastRow(); sheet.getRange(lastRow, 1).setHorizontalAlignment("center"); // วันที่ sheet.getRange(lastRow, 3).setHorizontalAlignment("center"); // เบอร์โทร sheet.getRange(lastRow, 4).setHorizontalAlignment("center"); // จังหวัด sheet.getRange(lastRow, 7).setHorizontalAlignment("center"); // อุณหภูมิ sheet.getRange(lastRow, 9).setHorizontalAlignment("center"); // รหัสคูปอง // ส่ง Response กลับไปยังหน้าเว็บว่าบันทึกเรียบร้อย return ContentService.createTextOutput(JSON.stringify({ result: "success", row: lastRow, timestamp: timestamp })).setMimeType(ContentService.MimeType.JSON); } catch (error) { // ดักจับข้อผิดพลาดและส่งรายงานกลับ return ContentService.createTextOutput(JSON.stringify({ result: "error", message: error.toString() })).setMimeType(ContentService.MimeType.JSON); } finally { // ปลดล็อกคิวเสมอ เพื่อให้รายการถัดไปทำงานต่อได้ lock.releaseLock(); } }
