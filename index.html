ตั้งค่าเชื่อมต่อ Google Sheets

บันทึกข้อมูลแบบสอบถามของลูกค้าลงชีตอัตโนมัติทันที

วาง Web App URL ที่ได้จาก Google Apps Script ที่นี่

โค้ด Apps Script เวอร์ชันใหม่ (ล็อกคิว & ล็อกเลข 0) 
/**
 * Carrier Cool Quest - Google Sheets Webhook Receiver (v2.0)
 * รองรับการบันทึกข้อมูลแบบสอบถาม, ล็อกคิวข้อมูล, และล็อกเลข 0 ของเบอร์โทร
 */

function doGet(e) {
  return ContentService.createTextOutput("Carrier Survey Webhook is active & ready!")
    .setMimeType(ContentService.MimeType.TEXT);
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  // ล็อกคิวรอไม่เกิน 15 วินาที เพื่อป้องกันข้อมูลชนกันกรณีส่งพร้อมกันหลายคน
  lock.tryLock(15000);

  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheetName = "Carrier_Survey_Data";
    var sheet = ss.getSheetByName(sheetName);

    // หากยังไม่มีแท็บนี้ ให้สร้างแท็บใหม่ให้อัตโนมัติ
    if (!sheet) {
      sheet = ss.insertSheet(sheetName);
    }

    // สร้างหัวตารางให้อัตโนมัติในครั้งแรก
    if (sheet.getLastRow() === 0) {
      var headers = [
        "วัน-เวลาบันทึก",
        "ชื่อ - นามสกุล",
        "เบอร์โทรศัพท์",
        "จังหวัด",
        "กิจกรรมวันหยุดที่ชอบ",
        "งานอดิเรก",
        "อุณหภูมิแอร์ในฝัน",
        "ของรางวัลที่ได้รับ",
        "รหัสคูปอง"
      ];
      sheet.appendRow(headers);
      
      // จัดรูปแบบหัวตารางสไตล์ Carrier
      var headerRange = sheet.getRange(1, 1, 1, headers.length);
      headerRange.setFontWeight("bold")
                 .setBackground("#002B66")
                 .setFontColor("#FFFFFF")
                 .setHorizontalAlignment("center");
      sheet.setFrozenRows(1);
    }

    var data = {};
    if (e.postData && e.postData.contents) {
      data = JSON.parse(e.postData.contents);
    } else if (e.parameter) {
      data = e.parameter;
    }

    var timestamp = Utilities.formatDate(new Date(), "GMT+7", "dd/MM/yyyy HH:mm:ss");
    
    // ใส่เครื่องหมาย ' นำหน้าเบอร์โทร เพื่อป้องกัน Google Sheets ตัดเลข 0 ตัวหน้าทิ้ง
    var rawPhone = data.phone ? String(data.phone).trim() : "-";
    var formattedPhone = (rawPhone !== "-" && !rawPhone.startsWith("'")) ? "'" + rawPhone : rawPhone;

    sheet.appendRow([
      timestamp,
      data.name || "-",
      formattedPhone,
      data.province || "-",
      data.weekend || "-",
      data.hobbies || "-",
      data.idealTemp || "-",
      data.prizeWon || "-",
      data.couponCode || "-"
    ]);

    return ContentService.createTextOutput(JSON.stringify({ "result": "success", "row": sheet.getLastRow() }))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ "result": "error", "error": err.message }))
      .setMimeType(ContentService.MimeType.JSON);

  } finally {
    lock.releaseLock();
  }
}
  1. เปิดไฟล์ Google Sheets ของคุณ แล้วไปที่ ส่วนขยาย (Extensions) > Apps Script
  2. ลบโค้ดเดิมออกทั้งหมด แล้ววางโค้ดชุดใหม่นี้ลงไป
  3. กดปุ่มสีน้ำเงิน ทำให้ใช้งานได้ (Deploy) > การทำให้ใช้งานได้ใหม่ (New deployment)
  4. ตรงเฟืองเลือก เว็บแอป (Web app), ตั้ง ใครมีสิทธิ์เข้าถึง (Who has access) เป็น ทุกคน (Anyone) เสมอ
  5. กด Deploy และคัดลอก URL ของเว็บแอป (ลงท้ายด้วย /exec) มาวางในช่องด้านบน
